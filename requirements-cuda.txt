# Common dependencies
-r requirements-common.txt

# torch wheels bundle their own CUDA runtime, so the cu124 wheels work on
# any host with a CUDA 12.x driver. ``uv pip install`` does NOT honour
# pyproject.toml's ``[tool.uv.sources]``, so the install command itself
# must select the cu124 wheel index, e.g.
#   uv pip install --torch-backend=cu124 -r requirements-cuda.txt
#   pip install --extra-index-url https://download.pytorch.org/whl/cu124 -r requirements-cuda.txt
torch == 2.5.1
torchvision == 0.20.1 # required for phi3v processor; updated alongside torch

# CUDA 12.4 toolkit components. The aarch64 torch wheel does NOT declare
# these as deps (it bundles its own runtime libs), so the resolver
# otherwise pulls whatever ``nvidia-*-cu12`` is newest (12.9 today).
# Pin them explicitly so the build sees a consistent 12.4 toolkit and
# nvcc/cicc/cudafe++ are available for compiling csrc.
# nccl 2.21.5 (the version torch cu124 ships on x86_64) has no aarch64
# wheel; 2.23.4 is the closest CUDA 12.4-compatible release that does.
# cudnn 9.1.0.70 likewise has no aarch64 wheel; 9.22.0.52 is forward-
# compatible with CUDA 12.4 and is what NVIDIA ships for ARM.
nvidia-cuda-nvcc-cu12 == 12.4.131
nvidia-cuda-runtime-cu12 == 12.4.127
nvidia-cuda-cupti-cu12 == 12.4.127
nvidia-cuda-nvrtc-cu12 == 12.4.127
nvidia-cublas-cu12 == 12.4.5.8
nvidia-cusparse-cu12 == 12.3.1.170
nvidia-cusolver-cu12 == 11.6.1.9
nvidia-curand-cu12 == 10.3.5.147
nvidia-cufft-cu12 == 11.2.1.3
nvidia-nccl-cu12 == 2.23.4
nvidia-nvtx-cu12 == 12.4.127
nvidia-nvjitlink-cu12 == 12.4.127
nvidia-cudnn-cu12 == 9.22.0.52

# Dependencies for NVIDIA GPUs
ray[default] >= 2.9
nvidia-ml-py >= 12.560.30 # for pynvml package

# SuperInfer's KV-swap path uses FlashInfer's paged-KV attention kernels.
# v0.2.9 is what the AE runs were validated against.
flashinfer-python == 0.2.9
